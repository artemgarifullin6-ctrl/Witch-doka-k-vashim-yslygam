Практическое освоение архитектурных подходов к конфигурированию периферийных узлов микроконтроллера STM32F411. == 2. Практическая реализация и этапы рефакторинга

=== 2.1. Шаг №1. Прямое битовое управление через регистры GPIO
Начальный этап разработки включал построение линейного алгоритма циклического переключения светодиодных индикаторов («бегущие огни»). Программа оперирует адресами периферии `GPIOA` и `GPIOC` непосредственно в теле главной функции `main()`.

.Исходный код первой программы (Процедурный подход)
[source,cpp]
----
#include "rccregisters.hpp" // for RCCм
#include "gpioaregisters.hpp" // for GPIOA
#include "gpiocregisters.hpp" // for GPIOC

std::uint32_t SystemCoreClock = 16'000'000U;

extern "C" {
    int __low_level_init(void)
    {
        RCC::CR::HSION::On::Set();
        while (RCC::CR::HSIRDY::NotReady::IsSet())
        {
        }
        RCC::CFGR::SW::Hsi::Set();
        while (!RCC::CFGR::SWS::Hsi::IsSet())
        {
        }
        RCC::APB2ENR::SYSCFGEN::Enable::Set();
        return 1;
    }
}

void delay(int cycles)
{
    for (int i = 0; i < cycles; ++i)
    {
        asm volatile("");
    }
}

int main()
{
    RCC::AHB1ENR::GPIOAEN::Enable::Set();
    RCC::AHB1ENR::GPIOCEN::Enable::Set();

    GPIOA::MODER::MODER5::Output::Set();
    GPIOC::MODER::MODER5::Output::Set();
    GPIOC::MODER::MODER8::Output::Set();
    GPIOC::MODER::MODER9::Output::Set();

    for (;;)
    {
        GPIOC::ODR::ODR5::High::Set();
        delay(1000000);
        GPIOC::ODR::ODR8::High::Set();
        delay(1000000);
        GPIOC::ODR::ODR9::High::Set();
        delay(1000000);
        GPIOA::ODR::ODR5::High::Set();
        delay(1000000);

        GPIOC::ODR::ODR5::Low::Set();
        delay(1000000);
        GPIOC::ODR::ODR8::Low::Set();
        delay(1000000);
        GPIOC::ODR::ODR9::Low::Set();
        delay(1000000);
        GPIOA::ODR::ODR5::Low::Set();
        delay(1000000);
    }
    return 1;
}
----

=== 2.2. Шаг №2. Наращивание функционала и усложнение процедурной модели
Для наглядного выявления системных недостатков монолитного кода в программу были внедрены дополнительные итерационные циклы. Светодиоды переводятся в режим работы с чередованием фаз активности (прогон одиночной волны с последующим сдвигом парных сигналов).

Расширение логики привело к перегрузке функции `main()` рутинными операциями и снижению прозрачности кода. Любое гипотетическое изменение аппаратной схемы (смена маски пинов или перенос линий на другой порт) привело бы к необходимости полной переработки алгоритмической составляющей.

.Исходный код второй программы (Расширенная процедурная логика)
[source,cpp]
----
#include "rccregisters.hpp" // for RCC
#include "gpioaregisters.hpp" // for GPIOA
#include "gpiocregisters.hpp" // for GPIOC

std::uint32_t SystemCoreClock = 16000000U;

extern "C" int __low_level_init(void)
{
    RCC::CR::HSION::On::Set();
    while (RCC::CR::HSIRDY::NotReady::IsSet())
    {
    }
    RCC::CFGR::SW::Hsi::Set();
    while (!RCC::CFGR::SWS::Hsi::IsSet())
    {
    }
    RCC::APB2ENR::SYSCFGEN::Enable::Set();
    return 1;
}

void delay(int cycles)
{
    for (int i = 0; i < cycles; ++i)
    {
        asm volatile ("");
    }
}

int main(void)
{
    RCC::AHB1ENR::GPIOAEN::Enable::Set();
    RCC::AHB1ENR::GPIOCEN::Enable::Set();

    GPIOA::MODER::MODER5::Output::Set();
    GPIOC::MODER::MODER5::Output::Set();
    GPIOC::MODER::MODER8::Output::Set();
    GPIOC::MODER::MODER9::Output::Set();

    while (1)
    {
        for (int count = 0; count < 3; ++count)
        {
            GPIOC::ODR::ODR5::High::Set();
            delay(1000000);
            GPIOC::ODR::ODR8::High::Set();
            delay(1000000);
            GPIOC::ODR::ODR9::High::Set();
            delay(1000000);
            GPIOA::ODR::ODR5::High::Set();
            delay(1000000);

            GPIOC::ODR::ODR5::Low::Set();
            delay(1000000);
            GPIOC::ODR::ODR8::Low::Set();
            delay(1000000);
            GPIOC::ODR::ODR9::Low::Set();
            delay(1000000);
            GPIOA::ODR::ODR5::Low::Set();
            delay(1000000);
        }

        for (int count = 0; count < 2; ++count)
        {
            GPIOA::ODR::ODR5::High::Set();
            delay(1000000);
            GPIOC::ODR::ODR8::High::Set();
            delay(1000000);

            GPIOA::ODR::ODR5::Low::Set();
            delay(1000000);
            GPIOC::ODR::ODR8::Low::Set();
            delay(1000000);

            GPIOA::ODR::ODR5::High::Set();
            delay(1000000);
            GPIOC::ODR::ODR9::High::Set();
            delay(1000000);

            GPIOC::ODR::ODR5::Low::Set();
            delay(1000000);
            GPIOC::ODR::ODR9::Low::Set();
            delay(1000000);
        }
    }
}

== 3. Проектирование компонентной архитектуры периферии
С целью декомпозиции проекта и устранения сильной связанности модулей была спроектирована новая объектная модель на основе предварительно составленной диаграммы UML.

Для структурирования исходных кодов в рабочей области IAR Workspace выделены две директории:
* *Contracts* — содержит интерфейсные описания (чисто виртуальные классы), накладывающие обязательства на сигнатуру методов.
* *Implementation* — включает программную реализацию логических блоков и аппаратных прослоек.

=== 3.1. Шаблонный класс Pin (Implementation/Pin.h)
Обобщенный класс `Pin` осуществляет конкретизацию методов абстракции `IPin`. Благодаря механизму метапрограммирования (`template`), конфигурация порта (`TPort`) и позиция вывода (`pinNum`) передаются как аргументы сборки.

Это позволяет компилятору разворачивать битовые сдвиги `1U << pinNum` непосредственно в коде во время сборки. В итоге генерируется максимально легковесный машинный код без создания тяжеловесных таблиц виртуальных методов времени выполнения (концепция Zero-cost abstractions).

[source,cpp]
----
#ifndef UNTITLED2_IMPLEMENTATION_PIN_H
#define UNTITLED2_IMPLEMENTATION_PIN_H

#include <cassert>
#include "IPin.h"

template <typename TPort, int pinNum>
class Pin final : public IPin
{
  public:
    void Set() override
    {
      TPort::Set(uint8_t(1U) << pinNum);
    }

    void Reset() override
    {
      //TPort::Reset(uint8_t(1U) << pinNum);
    }

    void Toggle() override
    {
      TPort::Toggle(uint8_t(1U) << pinNum);
    }

    bool GetState() const override
    {
      // Пока не реализуем (используется для кнопок)
      assert(true);
    }
};

#endif
----

=== 3.2. Интерфейс ILed (Contracts/ILed.h)
Задает единый программный контракт для управления абстрактной световой индикацией. Прикладные модули верхнего уровня взаимодействуют только с методами смены состояний `Light()`, `PutOut()` и `Toggle()`, сохраняя полную независимость от топологии платы.

[source,cpp]
----
#ifndef UNTITLED2_CONTRACTS_ILED_H
#define UNTITLED2_CONTRACTS_ILED_H

class ILed
{
public:
    virtual void Light() = 0;
    virtual void PutOut() = 0;
    virtual void Toggle() = 0;

    virtual ~ILed() = default;
};

#endif //UNTITLED2_CONTRACTS_ILED_H
----

=== 3.3. Класс Led (Implementation/Led.h)
Финальный класс `Led` реализует функционал индикатора через паттерн внедрения зависимостей, используя ссылку на интерфейсный объект `IPin&`. Конструктор связывает конкретный экземпляр светодиода с абстрактным аппаратным выводом. Данный подход гарантирует высокую гибкость: переназначение линий ввода-вывода происходит без изменения внутренней логики работы светодиодного компонента.

[source,cpp]
----
#ifndef UNTITLED2_IMPLEMENTATION_LED_H
#define UNTITLED2_IMPLEMENTATION_LED_H

#include "ILed.h"
#include "IPin.h"

class Led final : public ILed
{
public:
    explicit Led(IPin& pin)
        : pin(pin)
    {
    }

    void Light() override
    {
        pin.Set();
    }

    void PutOut() override
    {
        pin.Reset();
    }

    void Toggle() override
    {
        pin.Toggle();
    }

private:
    IPin& pin;
};

#endif //UNTITLED2_IMPLEMENTATION_LED_H
----

== 4. Тестирование, верификация и отладка
В ходе проверки и развёртывания разработанного программного обеспечения на стенде были зафиксированы следующие практические результаты:

1. *Сборка и прошивка процедурного кода:* Процедурные версии программ успешно скомпилированы в среде IAR. С помощью внутрисхемного программатора *ST-Link* бинарные файлы перенесены во флеш-память МК *STM32F411*. Тестирование первого варианта подтвердило циклическое переключение пинов `PC5` -> `PC8` -> `PC9` -> `PA5`. Проверка усложненного варианта подтвердила смену паттернов индикации: после 3 циклов поочередного включения стабильно воспроизводились 2 цикла парного мигания каналов (`PA5`/`PC8` и `PA5`/`PC9`).
2. *Проверка ООП-модели:* Логическая структура каталогов `Contracts` и `Implementation` была успешно развернута в проекте. Компилятор без предупреждений обработал заголовочные файлы `IPin.h`, `Pin.h`, `ILed.h` и `Led.h`.
3. *Анализ оптимизации:* Проведена верификация генерации кода для параметризованного шаблона `Pin`. Доказано, что вычисление масок пинов происходит на этапе компиляции, а быстродействие не уступает прямой процедурной записи.
